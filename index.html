    ISPNAT в сторону Интернета
    cat > /etc/nftables/nftables.nft << 'EOF'
#!/usr/sbin/nft -f
flush ruleset

table ip nat {
    chain postrouting {
        type nat hook postrouting priority srcnat; policy accept;
        oifname "enp0s3" masquerade
    }
}
EOF
systemctl enable --now nftables
systemctl restart nftables
nft list ruleset

    Динамическая маршрутизация OSPF

    FRR, area 0. Все интерфейсы пассивны, OSPF работает только на gre1. Аутентификация MD5.

    HQ-RTRFRR / ospfd
    sed -i 's/^ospfd=no/ospfd=yes/' /etc/frr/daemons

cat > /etc/frr/frr.conf << 'EOF'
frr defaults traditional
hostname hq-rtr.au-team.irpo
log syslog informational
!
interface gre1
 ip ospf network point-to-point
 ip ospf authentication message-digest
 ip ospf message-digest-key 1 md5 P@ssw0rd
 no ip ospf passive
exit
!
router ospf

    Динамическая трансляция адресов (NAT)

    Masquerade в сторону ISP. Трафик между офисами идёт через gre1 и не транслируется.

    HQ-RTRNAT → ISP (enp0s10)
    cat > /etc/nftables/nftables.nft << 'EOF'
#!/usr/sbin/nft -f
flush ruleset

table ip nat {
    chain postrouting {
        type nat hook postrouting priority srcnat; policy accept;
        oifname "enp0s10" masquerade
    }
}
EOF
