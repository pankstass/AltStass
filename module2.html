Задание 1

Контроллер домена Samba DC

Домен поднимается на BR-SRV с внутренним DNS Samba. HQ-CLI по-прежнему использует DNS HQ-SRV, поэтому на HQ-SRV добавляются SRV-записи, указывающие на контроллер. Kerberos чувствителен ко времени: при ошибке Clock skew сначала выполните задание 4.

BR-SRVУстановка и развёртывание домена
apt-get update && apt-get install -y task-samba-dc
grep -q 'br-srv.au-team.irpo' /etc/hosts || echo '10.1.100.10 br-srv.au-team.irpo br-srv' >> /etc/hosts
rm -f /etc/samba/smb.conf
rm -rf /var/lib/samba /var/cache/samba
mkdir -p /var/lib/samba/sysvol
samba-tool domain provision --realm=AU-TEAM.IRPO --domain=AU-TEAM \
  --server-role=dc --dns-backend=SAMBA_INTERNAL \
  --adminpass='P@ssw0rd' --option='dns forwarder = 10.0.100.20'
cp -f /var/lib/samba/private/krb5.conf /etc/krb5.conf
systemctl enable --now samba
printf 'search au-team.irpo\nnameserver 127.0.0.1\n' > /etc/net/ifaces/enp0s9.100/resolv.conf
systemctl restart network
samba-tool domain info 127.0.0.1
BR-SRVПолитика паролей, пользователи и группа hq
samba-tool domain passwordsettings set --min-pwd-length=8 \
  --account-lockout-threshold=5 --account-lockout-duration=30 \
  --reset-account-lockout-after=30
samba-tool group add hq
for i in $(seq 1 5); do
  samba-tool user add hquser$i 'P@ssw0rd'
  samba-tool group addmembers hq hquser$i
done
samba-tool group listmembers hq
samba-tool domain passwordsettings show
HQ-SRVSRV-записи контроллера в BIND
cat >> /var/lib/bind/zone/au-team.irpo.zone << 'EOF'
_ldap._tcp               IN SRV 0 100 389  br-srv.au-team.irpo.
_ldap._tcp.dc._msdcs     IN SRV 0 100 389  br-srv.au-team.irpo.
_kerberos._tcp           IN SRV 0 100 88   br-srv.au-team.irpo.
_kerberos._udp           IN SRV 0 100 88   br-srv.au-team.irpo.
_kerberos._tcp.dc._msdcs IN SRV 0 100 88   br-srv.au-team.irpo.
_kpasswd._tcp            IN SRV 0 100 464  br-srv.au-team.irpo.
_kpasswd._udp            IN SRV 0 100 464  br-srv.au-team.irpo.
_gc._tcp                 IN SRV 0 100 3268 br-srv.au-team.irpo.
EOF
sed -i 's/2027010101 ; serial/2027010102 ; serial/' /var/lib/bind/zone/au-team.irpo.zone
named-checkzone au-team.irpo /var/lib/bind/zone/au-team.irpo.zone
systemctl restart bind
dig @127.0.0.1 _ldap._tcp.au-team.irpo SRV +short
HQ-CLIВвод в домен
apt-get update && apt-get install -y task-auth-ad-sssd
system-auth write ad au-team.irpo hq-cli AU-TEAM 'administrator' 'P@ssw0rd'
reboot

Файловое хранилище RAID 0

Массив md0 из двух дополнительных дисков, раздел ext4, автомонтирование в /raid. Имена дисков проверьте по lsblk.

HQ-SRVmdadm, раздел, fstab
apt-get install -y mdadm
lsblk
mdadm --create /dev/md0 --level=0 --raid-devices=2 /dev/sdb /dev/sdc --run
mdadm --detail --scan > /etc/mdadm.conf
echo 'type=83' | sfdisk /dev/md0
mkfs.ext4 -F /dev/md0p1
mkdir -p /raid
echo '/dev/md0p1 /raid ext4 defaults 0 0' >> /etc/fstab
mount -a
df -h /raid
cat /proc/mdstat

Сервер NFS

Каталог /raid/nfs с доступом на чтение и запись только для сети HQ-CLI; на клиенте — автомонтирование в /mnt/nfs.

chmod 777 /raid/nfs
echo '/raid/nfs 10.0.200.0/24(rw,sync,no_subtree_check,no_root_squash)' >> /etc/exports
systemctl enable --now nfs-server

exportfs -v

apt-get install -y nfs-clients

echo '10.0.100.20:/raid/nfs /mnt/nfs nfs defaults,_netdev 0 0' >> /etc/fstab
mount -a
df -h /mnt/nfs
touch /mnt/nfs/test && ls -l /mnt/nfs

Служба времени chrony

ISP — сервер stratum 5 (minstratum 4 у вышестоящего источника даёт ровно 5). Клиенты обращаются к ближайшему адресу ISP, запросы проходят через NAT маршрутизаторов.

ISPNTP-сервер
apt-get install -y chrony
cat > /etc/chrony.conf << 'EOF'
server ntp1.vniiftri.ru iburst minstratum 4
local stratum 5
allow 172.16.1.0/28
allow 172.16.2.0/28
driftfile /var/lib/chrony/drift
makestep 1.0 3
rtcsync
EOF
systemctl enable --now chronyd
systemctl restart chronyd
chronyc tracking | grep -i stratum
HQ-SRVHQ-CLIКлиент (офис HQ)
apt-get install -y chrony
sed -i -E '/^(pool|server) /d' /etc/chrony.conf
echo 'server 172.16.1.1 iburst' >> /etc/chrony.conf
systemctl enable --now chronyd
systemctl restart chronyd
chronyc sources
BR-RTRBR-SRVКлиент (офис BR)
apt-get install -y chrony
sed -i -E '/^(pool|server) /d' /etc/chrony.conf
echo 'server 172.16.2.1 iburst' >> /etc/chrony.conf
systemctl enable --now chronyd
systemctl restart chronyd
chronyc sources

Ansible на BR-SRV

Подключение по ключам, чтобы ansible all -m ping отвечал pong без предупреждений. На HQ-CLI нужен sshd и пользователь.

HQ-CLIsshd и пользователь для Ansible
apt-get install -y openssh-server python3
id sshuser >/dev/null 2>&1 || useradd -m sshuser
echo 'sshuser:P@ssw0rd' | chpasswd
systemctl enable --now sshd
HQ-SRVHQ-RTRBR-RTRPython на управляемых машинах
apt-get install -y python3
BR-SRVКлючи SSH
apt-get install -y ansible
[ -f /root/.ssh/id_ed25519 ] || ssh-keygen -t ed25519 -N '' -f /root/.ssh/id_ed25519
ssh-copy-id -o StrictHostKeyChecking=accept-new -p 2027 sshuser@10.0.100.20
ssh-copy-id -o StrictHostKeyChecking=accept-new sshuser@10.0.200.30
ssh-copy-id -o StrictHostKeyChecking=accept-new net_admin@10.0.99.1
ssh-copy-id -o StrictHostKeyChecking=accept-new net_admin@10.1.99.1
BR-SRVИнвентарь и конфигурация
mkdir -p /etc/ansible
cat > /etc/ansible/hosts << 'EOF'
[hq]
hq-srv ansible_host=10.0.100.20 ansible_port=2027 ansible_user=sshuser
hq-cli ansible_host=10.0.200.30 ansible_user=sshuser
hq-rtr ansible_host=10.0.99.1 ansible_user=net_admin

[br]
br-rtr ansible_host=10.1.99.1 ansible_user=net_admin
EOF
cat > /etc/ansible/ansible.cfg << 'EOF'
[defaults]
inventory = /etc/ansible/hosts
host_key_checking = False
interpreter_python = auto_silent
EOF
cd /etc/ansible && ansible all -m ping

Статическая трансляция портов

Таблица nftables из задания 8 Модуля 1 дополняется цепочкой prerouting (DNAT). Файл перезаписывается целиком.

HQ-RTRDNAT на HQ-SRV
cat > /etc/nftables/nftables.nft << 'EOF'
#!/usr/sbin/nft -f
flush ruleset

table ip nat {
    chain prerouting {
        type nat hook prerouting priority dstnat; policy accept;
        iifname "enp0s10" tcp dport 8080 dnat to 10.0.100.20:80
        iifname "enp0s10" tcp dport 2027 dnat to 10.0.100.20:2027
    }

    chain postrouting {
        type nat hook postrouting priority srcnat; policy accept;
        oifname "enp0s10" masquerade
    }
}
EOF
systemctl enable --now nftables
systemctl restart nftables
nft list ruleset
BR-RTRDNAT на BR-SRV
cat > /etc/nftables/nftables.nft << 'EOF'
#!/usr/sbin/nft -f
flush ruleset

table ip nat {
    chain prerouting {
        type nat hook prerouting priority dstnat; policy accept;
        iifname "enp0s9" tcp dport 8080 dnat to 10.1.100.10:8080
        iifname "enp0s9" tcp dport 2027 dnat to 10.1.100.10:2027
    }

    chain postrouting {
        type nat hook postrouting priority srcnat; policy accept;
        oifname "enp0s9" masquerade
    }
}
EOF
systemctl enable --now nftables
systemctl restart nftables
nft list ruleset
