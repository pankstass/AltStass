Импорт пользователей в домен

Скрипт import.sh и users.csv лежат на Additional.iso. Файлы сохранены с переводами строк Windows (CRLF): без исправления bash падает с ошибкой «синтаксическая ошибка рядом с неожиданным маркером done». Образ смонтирован только для чтения, поэтому файлы копируются в /root/import.

BR-SRVИмпорт из users.csv✓ стенд
mkdir -p /opt/additional
# содержимое Additional.iso; если каталог пуст — монтируем диск
[ -n "$(ls -A /opt/additional 2>/dev/null)" ] || mount /dev/sr0 /opt/additional
mkdir -p /root/import
cp /opt/additional/import.sh /opt/additional/users.csv /root/import/
cd /root/import
# CRLF → LF, иначе «неожиданный маркер done»
sed -i 's/\r$//' import.sh users.csv
chmod +x import.sh
bash import.sh users.csv

Обратный прокси-сервер на ISP

Nginx на ISP принимает запросы к web.au-team.irpo и docker.au-team.irpo (обе записи в DNS HQ-SRV указывают на ISP). ISP не знает внутренних сетей офисов, поэтому запросы уходят на внешние адреса маршрутизаторов и дальше по DNAT из задания 8 Модуля 2. В ALT конфигурации лежат в sites-available.d / sites-enabled.d.

ISPNginx как reverse proxy✓ стенд
apt-get update && apt-get install -y nginx
systemctl enable --now nginx

cat > /etc/nginx/sites-available.d/revproxy.conf << 'EOF'
server {
    listen 80;
    server_name web.au-team.irpo;
    location / {
        proxy_pass http://172.16.1.2:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
server {
    listen 80;
    server_name docker.au-team.irpo;
    location / {
        proxy_pass http://172.16.2.2:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
EOF
ln -sf /etc/nginx/sites-available.d/revproxy.conf /etc/nginx/sites-enabled.d/revproxy.conf
rm -f /etc/nginx/sites-enabled.d/default.conf
nginx -t && systemctl restart nginx

Basic Auth в nginx только для web.au-team.irpo: пользователь WEB, пароль P@ssw0rd. Утилита htpasswd в ALT — пакет apache2-htpasswd. Файл пересобирается целиком, поэтому блок можно выполнять повторно.

ISPФайл паролей и auth_basic✓ стенд
apt-get install -y apache2-htpasswd
# -b — пароль в команде; без -b htpasswd спросит его дважды
htpasswd -cb /etc/nginx/.htpasswd WEB 'P@ssw0rd'

server {
    listen 80;
    server_name web.au-team.irpo;

    location / {
        proxy_pass http://172.16.1.2:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
server {
    listen 80;
    server_name docker.au-team.irpo;
    location / {
        proxy_pass http://172.16.2.2:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
EOF
ln -sf /etc/nginx/sites-available.d/revproxy.conf /etc/nginx/sites-enabled.d/revproxy.conf
rm -f /etc/nginx/sites-enabled.d/default.conf

Центр сертификации и HTTPS

ЦС на HQ-SRV (OpenSSL, RSA; на ДЭ с КриптоПро — ГОСТ). Сертификаты web и docker на 30 дней с SAN, ключи и сертификаты забираются на ISP через проброс SSH HQ-RTR (задание 8 Модуля 2), корневой сертификат — в доверенные на HQ-CLI. В ALT доверие добавляется через /etc/pki/ca-trust/source/anchors и update-ca-trust (не update-ca-certificates).

HQ-SRVКорневой ЦС и сертификаты web, docker✓ стенд
mkdir -p /etc/ssl/ca && cd /etc/ssl/ca
[ -f ca.key ] || openssl genrsa -out ca.key 4096
chmod 600 ca.key
openssl req -x509 -new -key ca.key -sha256 -days 30 -out ca.crt \
  -subj "/CN=AU-TEAM CA" \
  -addext "basicConstraints=critical,CA:TRUE" -addext "keyUsage=critical,keyCertSign,cRLSign"
for n in web docker; do
  openssl genrsa -out $n.key 2048
  openssl req -new -key $n.key -out $n.csr -subj "/CN=$n.au-team.irpo"
  printf 'subjectAltName=DNS:%s.au-team.irpo\nextendedKeyUsage=serverAuth\n' $n > $n.ext
  openssl x509 -req -in $n.csr -CA ca.crt -CAkey ca.key -CAcreateserial \
    -out $n.crt -days 30 -sha256 -extfile $n.ext
done
openssl x509 -in web.crt -noout -subject -issuer -dates -ext subjectAltName
openssl verify -CAfile ca.crt web.crt docker.crt
Ожидается
web.crt: OK, docker.crt: OK; notAfter — через 30 дней
HQ-SRVВыложить файлы для ISP и HQ-CLI
install -d -o sshuser -g sshuser -m 700 /home/sshuser/certs
cp /etc/ssl/ca/{ca.crt,web.crt,web.key,docker.crt,docker.key} /home/sshuser/certs/
chown sshuser:sshuser /home/sshuser/certs/*
ls -l /home/sshuser/certs
ISPСертификаты и HTTPS в nginx✓ стенд
mkdir -p /etc/nginx/ssl && cd /etc/nginx/ssl
# через DNAT HQ-RTR (172.16.1.2:2027 → HQ-SRV)
scp -P 2027 -o StrictHostKeyChecking=accept-new 'sshuser@172.16.1.2:certs/*' .
chown root:root /etc/nginx/ssl/*
chmod 600 /etc/nginx/ssl/*.key

cat > /etc/nginx/sites-available.d/revproxy.conf << 'EOF'
server {
    listen 80;
    server_name web.au-team.irpo docker.au-team.irpo;
    return 301 https://$host$request_uri;
}
server {
    listen 443 ssl;
    server_name web.au-team.irpo;
    ssl_certificate     /etc/nginx/ssl/web.crt;
    ssl_certificate_key /etc/nginx/ssl/web.key;
    auth_basic "Authorized access only";
    auth_basic_user_file /etc/nginx/.htpasswd;
    location / {
        proxy_pass http://172.16.1.2:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
server {
    listen 443 ssl;
    server_name docker.au-team.irpo;
    ssl_certificate     /etc/nginx/ssl/docker.crt;
    ssl_certificate_key /etc/nginx/ssl/docker.key;
    location / {
        proxy_pass http://172.16.2.2:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
EOF
ln -sf /etc/nginx/sites-available.d/revproxy.conf /etc/nginx/sites-enabled.d/revproxy.conf
rm -f /etc/nginx/sites-enabled.d/default.conf
nginx -t && systemctl restart nginx

cat /etc/default/atop

Инвентаризация через Ansible

Плейбук на BR-SRV собирает имя и IP HQ-SRV и HQ-CLI и сохраняет их на BR-SRV в /etc/ansible/PC-INFO/<хост>.yml — для этого задача делегируется на localhost. Использует инвентарь из задания 5 Модуля 2.

BR-SRVПлейбук inventory.yml✓ стенд
mkdir -p /etc/ansible/PC-INFO /etc/ansible/playbook
cat > /etc/ansible/playbook/inventory.yml << 'EOF'
- name: Инвентаризация HQ-SRV и HQ-CLI
  hosts: hq-srv,hq-cli
  gather_facts: yes
  tasks:
    - name: Сохранение имени и IP на BR-SRV
      copy:
        content: |
          hostname: {{ ansible_fqdn }}
          ip_address: {{ ansible_default_ipv4.address }}
        dest: "/etc/ansible/PC-INFO/{{ inventory_hostname }}.yml"
      delegate_to: localhost
EOF
ansible-playbook /etc/ansible/playbook/inventory.yml

cat /etc/ansible/PC-INFO/*.yml

Защита SSH: fail2ban

Jail sshd на порту 2027 (не 22!), 3 попытки, блокировка 60 с (в новых версиях bantime — в секундах). В ALT журнал sshd читается через systemd (backend = systemd, нужен python3-module-systemd), блокировка — через nftables.

HQ-SRVУстановка и jail.local✓ стенд
apt-get install -y fail2ban python3-module-systemd nftables
cat > /etc/fail2ban/jail.local << 'EOF'
[DEFAULT]
backend   = systemd
banaction = nftables-multiport
ignoreip  = 127.0.0.1/8

[sshd]
enabled  = true
port     = 2027
filter   = sshd
maxretry = 3
findtime = 600
bantime  = 60
EOF
systemctl enable --now fail2ban
systemctl restart fail2ban
fail2ban-client status sshd

Резервное копирование BorgBackup

Аналог «Кибер Бэкап» с защитой от шифровальщиков: HQ-SRV отправляет /etc в репозиторий на HQ-CLI. Ключ HQ-SRV на HQ-CLI ограничен командой borg serve --append-only --restrict-to-path /backup: заражённый HQ-SRV может добавлять архивы, но не удалять и не перезаписывать старые. Репозиторий зашифрован (repokey), копия — ежедневно по таймеру systemd.

HQ-SRVHQ-CLIПакет borg✓ стенд
apt-get install -y borg
borg --version
HQ-CLIПользователь borguser и каталог /backup✓ стенд
id borguser >/dev/null 2>&1 || useradd -m -s /bin/bash borguser
mkdir -p /backup && chown borguser:borguser /backup && chmod 700 /backup
# если в sshd задан AllowUsers — добавить borguser
if grep -q '^AllowUsers' /etc/openssh/sshd_config && ! grep -q '^AllowUsers.*borguser' /etc/openssh/sshd_config; then
  sed -i '/^AllowUsers/ s/$/ borguser/' /etc/openssh/sshd_config
  sshd -t && systemctl restart sshd
fi
HQ-SRVКлюч SSH и пароль репозитория✓ стенд
[ -f /root/.ssh/borg_key ] || ssh-keygen -t ed25519 -N '' -f /root/.ssh/borg_key
echo 'P@ssw0rd' > /root/.borg-pass && chmod 600 /root/.borg-pass
scp -P 2027 /root/.ssh/borg_key.pub sshuser@10.0.200.30:/tmp/
HQ-CLIКлюч только для borg serve --append-only, репозиторий✓ стенд
mkdir -p /home/borguser/.ssh
grep -qF "$(cut -d' ' -f2 /tmp/borg_key.pub)" /home/borguser/.ssh/authorized_keys 2>/dev/null || \
  echo "command=\"borg serve --append-only --restrict-to-path /backup\",restrict $(cat /tmp/borg_key.pub)" >> /home/borguser/.ssh/authorized_keys
chown -R borguser:borguser /home/borguser/.ssh
chmod 700 /home/borguser/.ssh && chmod 600 /home/borguser/.ssh/authorized_keys
[ -d /backup/hq-srv ] || sudo -u borguser BORG_PASSPHRASE='P@ssw0rd' borg init --encryption=repokey /backup/hq-srv
HQ-SRVСкрипт и ежедневный таймер✓ стенд
cat > /usr/local/bin/backup-etc.sh << 'EOF'
#!/bin/bash
export BORG_RSH='ssh -i /root/.ssh/borg_key -p 2027 -o StrictHostKeyChecking=accept-new -o IdentitiesOnly=yes'
export BORG_PASSCOMMAND='cat /root/.borg-pass'
REPO='ssh://borguser@10.0.200.30:2027/backup/hq-srv'
borg create --stats --compression lz4 "$REPO::etc-{now:%Y-%m-%d_%H%M}" /etc
EOF
chmod 700 /usr/local/bin/backup-etc.sh
cat > /etc/systemd/system/backup-etc.service << 'EOF'
[Unit]
Description=Backup /etc to HQ-CLI (borg)
After=network-online.target
Wants=network-online.target
[Service]
Type=oneshot
ExecStart=/usr/local/bin/backup-etc.sh
